using System;
using System.IO;
using System.Windows.Forms;

namespace RockPaperScissorsGame
{
    public partial class AdminLoginForm : Form
    {
        private const string AdminFilePath = "admin.txt";

        public AdminLoginForm()
        {
            InitializeComponent();
            EnsureAdminFileExists();
            this.KeyPreview = true; // Дозволяє обробляти натискання клавіш для всієї форми
            this.KeyDown += AdminLoginForm_KeyDown; // Додаємо обробник для клавіш

            // Додаємо підказки до текстових полів
            AddToolTips();
        }

        private void EnsureAdminFileExists()
        {
            if (!File.Exists(AdminFilePath))
            {
                File.WriteAllText(AdminFilePath, "Login: admin\nPassword: admin");
            }
        }

        private void AddToolTips()
        {
            var toolTip = new ToolTip();
            toolTip.SetToolTip(txtUsername, "За замовчуванням Логін: admin");
            toolTip.SetToolTip(txtPassword, "За замовчуванням Пароль: admin");
        }

        private void btnLogin_Click(object sender, EventArgs e)
        {
            string username = txtUsername.Text;
            string password = txtPassword.Text;

            try
            {
                if (File.Exists(AdminFilePath))
                {
                    var adminData = File.ReadAllLines(AdminFilePath);
                    if (adminData.Length < 2)
                    {
                        MessageBox.Show("Файл даних адміністратора пошкоджений!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                        return;
                    }

                    string savedUsername = adminData[0].Split(':')[1].Trim();
                    string savedPassword = adminData[1].Split(':')[1].Trim();

                    if (username == savedUsername && password == savedPassword)
                    {
                        AdminMenuForm adminMenu = new AdminMenuForm();
                        adminMenu.ShowDialog();
                        this.Close();
                    }
                    else
                    {
                        MessageBox.Show("Невірний логін або пароль!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    }
                }
                else
                {
                    MessageBox.Show("Файл адміністратора не знайдено!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Виникла помилка: {ex.Message}", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnCancel_Click(object sender, EventArgs e)
        {
            this.Close();
        }

        private void AdminLoginForm_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Enter)
            {
                btnLogin_Click(sender, e);
            }
            else if (e.KeyCode == Keys.Escape)
            {
                btnCancel_Click(sender, e);
            }
        }
    }
}

using System;
using System.IO;
using System.Windows.Forms;

namespace RockPaperScissorsGame
{
    public partial class ChangeAdminCredentialsForm : Form
    {
        private const string AdminFilePath = "admin.txt";

        public ChangeAdminCredentialsForm()
        {
            InitializeComponent();
            this.KeyPreview = true; // Дозволяє обробку клавіш для всієї форми
            this.KeyDown += ChangeAdminCredentialsForm_KeyDown;
        }

        private void btnSave_Click(object sender, EventArgs e)
        {
            string newUsername = txtNewEmail.Text.Trim();
            string newPassword = txtNewPassword.Text.Trim();
            string confirmPassword = txtConfirmPassword.Text.Trim();

            if (string.IsNullOrWhiteSpace(newUsername) || string.IsNullOrWhiteSpace(newPassword))
            {
                MessageBox.Show("Будь ласка, заповніть всі поля!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            if (newPassword != confirmPassword)
            {
                MessageBox.Show("Паролі не співпадають!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            try
            {
                if (File.Exists(AdminFilePath))
                {
                    // Зчитуємо поточні дані адміністратора
                    var adminData = File.ReadAllLines(AdminFilePath);
                    if (adminData.Length >= 2)
                    {
                        string currentLogin = adminData[0].Split(':')[1].Trim();
                        string currentPassword = adminData[1].Split(':')[1].Trim();

                        // Перевірка, чи не збігаються нові дані з поточними
                        if (newUsername == currentLogin && newPassword == currentPassword)
                        {
                            MessageBox.Show("Неможливо змінити дані на такі ж самі!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                            return;
                        }
                    }

                    // Оновлюємо дані адміністратора
                    File.WriteAllText(AdminFilePath, $"Login: {newUsername}\nPassword: {newPassword}");
                    MessageBox.Show("Облікові дані адміністратора успішно змінено!", "Інформація", MessageBoxButtons.OK, MessageBoxIcon.Information);
                    this.Close();
                }
                else
                {
                    MessageBox.Show("Файл облікових записів не знайдено!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Сталася помилка: {ex.Message}", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnCancel_Click(object sender, EventArgs e)
        {
            this.Close();
        }

        private void ChangeAdminCredentialsForm_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Enter)
            {
                btnSave_Click(sender, e);
            }
            else if (e.KeyCode == Keys.Escape)
            {
                btnCancel_Click(sender, e);
            }
        }
    }
}

using System;
using System.IO;
using System.Linq;
using System.Windows.Forms;

namespace RockPaperScissorsGame
{
    public partial class ChangeUserCredentialsForm : Form
    {
        private readonly string currentUserEmail;

        public ChangeUserCredentialsForm(string email)
        {
            InitializeComponent();
            currentUserEmail = email;

            // Дозволяємо перехоплювати натискання клавіш для всієї форми
            this.KeyPreview = true;
            this.KeyDown += ChangeUserCredentialsForm_KeyDown;
        }

        private void btnSave_Click(object sender, EventArgs e)
        {
            string newName = txtName.Text.Trim();
            string newEmail = txtEmail.Text.Trim();
            string newPassword = txtPassword.Text.Trim();

            if (string.IsNullOrWhiteSpace(newName) || string.IsNullOrWhiteSpace(newEmail) || string.IsNullOrWhiteSpace(newPassword))
            {
                MessageBox.Show("Усі поля повинні бути заповнені!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            try
            {
                if (File.Exists("accounts.txt"))
                {
                    var accounts = File.ReadAllLines("accounts.txt").ToList();
                    int index = accounts.FindIndex(a => a.Contains(currentUserEmail));
                    if (index != -1)
                    {
                        accounts[index] = $"{newName},{newEmail},{newPassword}";
                        File.WriteAllLines("accounts.txt", accounts);
                        MessageBox.Show("Дані успішно змінено!", "Інформація", MessageBoxButtons.OK, MessageBoxIcon.Information);
                        this.DialogResult = DialogResult.OK;
                        this.Close();
                    }
                    else
                    {
                        MessageBox.Show("Користувача не знайдено!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                    }
                }
                else
                {
                    MessageBox.Show("Файл облікових записів не знайдено!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Помилка: {ex.Message}", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnCancel_Click(object sender, EventArgs e)
        {
            this.Close();
        }

        private void ChangeUserCredentialsForm_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Enter)
            {
                btnSave.PerformClick(); // Виклик функції збереження змін
            }
            else if (e.KeyCode == Keys.Escape)
            {
                btnCancel.PerformClick(); // Виклик функції скасування
            }
        }
    }
}

DataManager.cs --- 

using System;
using System.Collections.Generic;
using System.IO;
using System.Windows.Forms;

namespace RockPaperScissorsGame
{
    public static class DataManager
    {
        private const string AccountsFilePath = "accounts.txt";
        private const string StatisticsFolderPath = "statistics";

        static DataManager()
        {
            if (!Directory.Exists(StatisticsFolderPath))
            {
                Directory.CreateDirectory(StatisticsFolderPath);
            }
        }

        public static bool AddUser(string name, string email, string password)
        {
            if (string.IsNullOrWhiteSpace(name) || string.IsNullOrWhiteSpace(email) || string.IsNullOrWhiteSpace(password))
            {
                return false;
            }

            if (IsUserRegistered(email))
            {
                return false;
            }

            File.AppendAllText(AccountsFilePath, $"{name},{email},{password}\n");
            return true;
        }

        public static bool IsUserRegistered(string email)
        {
            if (!File.Exists(AccountsFilePath))
            {
                return false;
            }

            var accounts = File.ReadAllLines(AccountsFilePath);
            foreach (var account in accounts)
            {
                var parts = account.Split(',');
                if (parts.Length > 1 && parts[1].Equals(email, StringComparison.OrdinalIgnoreCase))
                {
                    return true;
                }
            }

            return false;
        }

        public static string GetUserName(string? email)
        {
            if (email == "Комп'ютер")
            {
                return "Комп'ютер";
            }

            if (string.IsNullOrEmpty(email))
            {
                return "Невідомий гравець";
            }

            if (File.Exists(AccountsFilePath))
            {
                var accounts = File.ReadAllLines(AccountsFilePath);
                foreach (var account in accounts)
                {
                    var parts = account.Split(',');
                    if (parts.Length > 1 && parts[1].Equals(email, StringComparison.OrdinalIgnoreCase))
                    {
                        return parts[0]; // Повертаємо ім'я
                    }
                }
            }

            return "Невідомий гравець";
        }

        public static void AddGameResult(string userEmail, string opponentName, string result, string playerChoice, string opponentChoice)
        {
            try
            {
                string filePath = Path.Combine(StatisticsFolderPath, $"{userEmail}.statistic.txt");

                if (!Directory.Exists(StatisticsFolderPath))
                {
                    Directory.CreateDirectory(StatisticsFolderPath);
                }

                string userName = GetUserName(userEmail);

                string opponentDisplayName = opponentName == "Комп'ютер" ? "Комп'ютер" : GetUserName(opponentName);

                string logEntry = $"{DateTime.Now}: {userName} грав проти {opponentDisplayName}, Ваш хід: {playerChoice}, Хід супротивника: {opponentChoice}, результат: {result}";

                File.AppendAllText(filePath, logEntry + Environment.NewLine);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Помилка запису статистики: {ex.Message}", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        public static List<string> GetStatistics(string email)
        {
            var results = new List<string>();
            string filePath = Path.Combine(StatisticsFolderPath, $"{email}.statistic.txt");

            if (File.Exists(filePath))
            {
                results.AddRange(File.ReadAllLines(filePath));
            }

            return results;
        }

        public static void ClearStatistics(string email)
        {
            try
            {
                string statisticsFilePath = Path.Combine(StatisticsFolderPath, $"{email}.statistic.txt");

                if (File.Exists(statisticsFilePath))
                {
                    File.Delete(statisticsFilePath);
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Помилка при видаленні статистики для {email}: {ex.Message}");
            }
        }

        public static void RemoveUser(string email)
        {
            // Удаляем пользователя из accounts.txt
            if (File.Exists(AccountsFilePath))
            {
                var accounts = File.ReadAllLines(AccountsFilePath);
                var filteredAccounts = new List<string>();

                foreach (var account in accounts)
                {
                    var parts = account.Split(',');
                    if (parts.Length > 1 && !parts[1].Equals(email, StringComparison.OrdinalIgnoreCase))
                    {
                        filteredAccounts.Add(account);
                    }
                }

                File.WriteAllLines(AccountsFilePath, filteredAccounts);
            }

            ClearStatistics(email);
        }
    }

        private async Task HandlePlayerChoice(string choice)
        {
            try
            {
                playerChoice = choice;
                playerChoiceBox.Image = Image.FromFile($"Resources/{choice}.jpg");

                await Task.Delay(500); // Затримка перед вибором комп'ютера
                computerChoice = GetComputerChoice();
                computerChoiceBox.Image = Image.FromFile($"Resources/{computerChoice}.jpg");

                await Task.Delay(500); // Затримка перед відображенням результату
                ShowResult();

                // Після показу результату скидаємо стан гри
                ResetGame();

            {
                MessageBox.Show($"Не вдалося знайти файл зображення: {ex.Message}", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);

            DataManager.AddGameResult(playerEmail, "Комп'ютер", result, TranslateChoice(playerChoice ?? string.Empty), TranslateChoice(computerChoice ?? string.Empty));
        }

using System;
using System.IO;
using System.Linq;
using System.Windows.Forms;

namespace RockPaperScissorsGame
{
    public partial class PlayerManagementForm : Form
    {
        public PlayerManagementForm()
        {
            InitializeComponent();
            LoadPlayers();
            this.KeyPreview = true; // Дозволяє обробляти натискання клавіш для всієї форми
            this.KeyDown += PlayerManagementForm_KeyDown; // Додаємо обробник для клавіш
        }

        private void LoadPlayers()
        {
            lstPlayers.Items.Clear();
            if (File.Exists("accounts.txt"))
            {
                var accounts = File.ReadAllLines("accounts.txt");
                foreach (var account in accounts)
                {
                    var parts = account.Split(',');
                    if (parts.Length > 0)
                    {
                        lstPlayers.Items.Add(parts[0]); // Додаємо email гравця
                    }
                }
            }
        }

        private void btnDeletePlayer_Click(object sender, EventArgs e)
        {
            if (lstPlayers.SelectedItem != null)
            {
                string selectedPlayerName = lstPlayers.SelectedItem.ToString();

                // Отримуємо email гравця з accounts.txt
                var accounts = File.ReadAllLines("accounts.txt");
                string? emailToDelete = accounts
                    .Select(line => line.Split(','))
                    .FirstOrDefault(parts => parts[0] == selectedPlayerName)?
                    .ElementAtOrDefault(1);

                if (!string.IsNullOrEmpty(emailToDelete))
                {
                    DataManager.RemoveUser(emailToDelete); // Видалення користувача і його статистики
                    MessageBox.Show("Гравця та його статистику успішно видалено!", "Інформація", MessageBoxButtons.OK, MessageBoxIcon.Information);
                    LoadPlayers(); // Оновлення списку гравців
                }
                else
                {
                    MessageBox.Show("Не вдалося знайти email для видалення.", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                }
            }
            else
            {
                MessageBox.Show("Будь ласка, виберіть гравця для видалення.", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
            }
        }

        private void btnClose_Click(object sender, EventArgs e)
        {
            this.Close();
        }

        private void PlayerManagementForm_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Escape)
            {
                btnClose_Click(sender, e); // Виклик функції для кнопки "Закрити"
            }
        }
    }
}

        private void StatisticsForm_Load(object sender, EventArgs e)
        {
            LoadStatistics();

        private void LoadStatistics()
        {
            try
            {
                lstStatistics.Items.Clear();
                string statisticsFilePath = Path.Combine("statistics", $"{userEmail}.statistic.txt");

                if (File.Exists(statisticsFilePath))
                {
                    string[] lines = File.ReadAllLines(statisticsFilePath);

                    if (lines.Any())
                    {
                        lstStatistics.Items.AddRange(lines);
                    }
                    else
                    {
                        lstStatistics.Items.Add("Немає записів у статистиці.");
                    }
                }
                else
                {
                    MessageBox.Show("Файл статистики не знайдено!", "Інформація", MessageBoxButtons.OK, MessageBoxIcon.Information);
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Помилка завантаження статистики: {ex.Message}", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnClearStatistics_Click(object sender, EventArgs e)
        {
            try
            {
                DataManager.ClearStatistics(userEmail);
                lstStatistics.Items.Clear();
                MessageBox.Show("Статистика успішно очищена!", "Інформація", MessageBoxButtons.OK, MessageBoxIcon.Information);

            catch (Exception ex)
            {
                MessageBox.Show($"Помилка очищення статистики: {ex.Message}", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);

        }

        private void btnClearSelectedStat_Click(object sender, EventArgs e)

using System;
using System.IO;
using System.Text.RegularExpressions;
using System.Windows.Forms;

namespace RockPaperScissorsGame
{
    public partial class UserLoginForm : Form
    {
        public string UserEmail { get; private set; }
        public string UserPassword { get; private set; }

        public UserLoginForm()
        {
            InitializeComponent();
            this.KeyPreview = true; // Дозволяє обробляти натискання клавіш для всієї форми
            this.KeyDown += UserLoginForm_KeyDown; // Додаємо обробник для клавіш
            this.txtPassword.KeyDown += new KeyEventHandler(txtPassword_KeyDown); // Додаємо обробник для TextBox
        }

        private bool ValidateUser(string email, string password)
        {
            if (!File.Exists("accounts.txt"))
            {
                MessageBox.Show("Файл accounts.txt не знайдено!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return false;
            }

            var accounts = File.ReadAllLines("accounts.txt");
            bool emailFound = false;

            foreach (var account in accounts)
            {
                var parts = account.Split(',');
                if (parts.Length == 3 && parts[1].Equals(email, StringComparison.OrdinalIgnoreCase))
                {
                    emailFound = true;
                    if (parts[2] == password)
                    {
                        return true;
                    }
                    else
                    {
                        MessageBox.Show("Невірний пароль!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                        return false;
                    }
                }
            }

            if (!emailFound)
            {
                MessageBox.Show("Користувача з таким email не знайдено!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }

            return false;
        }

        private void btnLogin_Click(object sender, EventArgs e)
        {
            string email = txtEmail.Text.Trim();
            string password = txtPassword.Text.Trim();

            if (string.IsNullOrEmpty(email) || string.IsNullOrEmpty(password))
            {
                MessageBox.Show("Будь ласка, заповніть усі поля!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            // Перевірка формату email
            if (!Regex.IsMatch(email, @"^[a-zA-Z0-9]+@[a-zA-Z]+\.[a-zA-Z]{2,}$"))
            {
                MessageBox.Show("Невірний формат email!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            if (!ValidateUser(email, password))
            {
                return;
            }

            // Якщо користувач існує, закриваємо форму з позитивним результатом
            UserEmail = email;
            this.DialogResult = DialogResult.OK;
            this.Close();
        }

        private void btnCancel_Click(object sender, EventArgs e)
        {
            this.DialogResult = DialogResult.Cancel;
            this.Close();
        }

        private void UserLoginForm_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Enter)
            {
                btnLogin.PerformClick(); // Виклик функції входу
            }
            else if (e.KeyCode == Keys.Escape)
            {
                btnCancel.PerformClick(); // Виклик функції скасування
            }
        }

        private void txtPassword_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Enter)
            {
                btnLogin.PerformClick(); // Виклик функції входу
            }
        }
    }

using System;
using System.IO;
using System.Text.RegularExpressions;
using System.Windows.Forms;

namespace RockPaperScissorsGame
{
    public partial class UserRegistrationForm : Form
    {
        public UserRegistrationForm()
        {
            InitializeComponent();
            this.KeyPreview = true; // Дозволяє перехоплювати натискання клавіш на рівні форми
            this.KeyDown += UserRegistrationForm_KeyDown; // Додаємо обробник подій клавіш
        }

        private void btnRegister_Click(object sender, EventArgs e)
        {
            string name = txtName.Text.Trim();
            string email = txtEmail.Text.Trim();
            string password = txtPassword.Text;

            if (string.IsNullOrWhiteSpace(name) || string.IsNullOrWhiteSpace(email) || string.IsNullOrWhiteSpace(password))
            {
                MessageBox.Show("Будь ласка, заповніть всі поля!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            if (!Regex.IsMatch(email, @"^[a-zA-Z0-9]+@[a-zA-Z]+\.[a-zA-Z]{2,}$"))
            {
                MessageBox.Show("Введено некоректний email!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            if (UserExists(email))
            {
                MessageBox.Show("Користувач з таким email вже існує!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
                return;
            }

            if (DataManager.AddUser(name, email, password))
            {
                this.DialogResult = DialogResult.OK;
                this.Close();
            }
            else
            {
                MessageBox.Show("Сталася помилка при реєстрації користувача!", "Помилка", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void btnCancel_Click(object sender, EventArgs e)
        {
            this.DialogResult = DialogResult.Cancel;
            this.Close();
        }

        private bool UserExists(string email)
        {
            if (!File.Exists("accounts.txt"))
            {
                return false;
            }

            var accounts = File.ReadAllLines("accounts.txt");
            foreach (var account in accounts)
            {
                var parts = account.Split(',');
                if (parts.Length >= 2 && parts[1].Equals(email, StringComparison.OrdinalIgnoreCase))
                {
                    return true;
                }
            }

            return false;
        }

        private void UserRegistrationForm_KeyDown(object sender, KeyEventArgs e)
        {
            if (e.KeyCode == Keys.Enter)
            {
                btnRegister.PerformClick(); // Емуляція натискання кнопки "Реєстрація"
            }
            else if (e.KeyCode == Keys.Escape)
            {
                btnCancel.PerformClick(); // Емуляція натискання кнопки "Скасувати"
            }
        }
    }
}
